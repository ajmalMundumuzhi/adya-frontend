import axios from "axios";

const api = axios.create({
  baseURL: import.meta.env.VITE_API_URL || "http://localhost:5000/api",
  headers: {
    "Content-Type": "application/json",
  },
  timeout: 15000,

api.interceptors.request.use((config) => {
  const token = localStorage.getItem("adya-customer-token");

  if (token) {
    config.headers.Authorization = `Bearer ${token}`;
  }

  return config;
});

import api from "./api";

const TOKEN_KEY = "adya-customer-token";
const USER_KEY = "adya-customer-user";

function saveSession(data) {
  if (!data.token || !data.user) {
    throw new Error("Authentication response is incomplete");
  }

  localStorage.setItem(TOKEN_KEY, data.token);
  localStorage.setItem(USER_KEY, JSON.stringify(data.user));

  window.dispatchEvent(new Event("adya-auth-change"));

  return data.user;
}

export function logoutCustomer() {
  localStorage.removeItem(TOKEN_KEY);
  localStorage.removeItem(USER_KEY);

  window.dispatchEvent(new Event("adya-auth-change"));
}

export async function loginCustomer(credentials) {
  const response = await api.post("/users/login", credentials);
  return saveSession(response.data);
}

export async function registerCustomer(data) {
  const response = await api.post("/users/register", data);
  return saveSession(response.data);
}

export function getCurrentCustomer() {
  try {
    const user = localStorage.getItem(USER_KEY);
    return user ? JSON.parse(user) : null;
  } catch {
    return null;
  }
}

export function getCustomerToken() {
  return localStorage.getItem(TOKEN_KEY);
}

import { useEffect, useState } from 'react';
import { Link } from 'react-router-dom';
import { Heart, ShoppingBag, Trash2, ArrowRight } from 'lucide-react';
import useCurrency from '../hooks/useCurrency';

const WISHLIST_KEY = 'adya-wishlist';

function readWishlist() {
  try {
    const saved = localStorage.getItem(WISHLIST_KEY);
    return saved ? JSON.parse(saved) : [];
  } catch {
    return [];
  }
}

export default function Wishlist() {
  const [products, setProducts] = useState(readWishlist);
  const { format } = useCurrency();

  useEffect(() => {
    try {
      localStorage.setItem(WISHLIST_KEY, JSON.stringify(products));
    } catch {
        console.log("Error storing.")
    }
  }, [products]);

  function removeProduct(productId) {
    setProducts((current) =>
      current.filter((product) => product.id !== productId),
    );
  }

  function moveToCart(product) {
    try {
      const saved = localStorage.getItem('adya-cart');
      const cart = saved ? JSON.parse(saved) : [];
      const existing = cart.find((line) => line.productId === product.id);

      let updatedCart;

      if (existing) {
        updatedCart = cart.map((line) =>
          line.productId === product.id
            ? { ...line, quantity: Math.min(9, line.quantity + 1), product }
            : line,
        );
      } else {
        updatedCart = [
          ...cart,
          { productId: product.id, quantity: 1, product },
        ];
      }

      localStorage.setItem('adya-cart', JSON.stringify(updatedCart));
      removeProduct(product.id);
    } catch {
      return;
    }
  }

  return (
    <main className="min-h-screen bg-ivory-100 px-4 py-12 sm:px-6 sm:py-16 lg:px-8">
      <div className="mx-auto max-w-7xl">
        <div className="mb-10 border-b border-[#C4B382]/40 pb-7 sm:mb-12">
          <p className="mb-3 text-[10px] font-semibold uppercase tracking-[0.25em] text-[#8D7547]">
            Your personal collection
          </p>
          <div className="flex items-center gap-3">
            <Heart size={27} strokeWidth={1.4} className="text-[#A58F5D]" />
            <h1 className="font-display text-4xl text-[#51452F] sm:text-5xl">
              My Wishlist
            </h1>
          </div>
          <p className="mt-3 text-sm text-ink/60">
            {products.length} {products.length === 1 ? 'piece' : 'pieces'} saved for later
          </p>
        </div>

        {products.length === 0 ? (
          <div className="flex min-h-[40vh] flex-col items-center justify-center py-16 text-center">
            <Heart
              size={48}
              strokeWidth={1.2}
              className="text-[#C4B382]"
            />
            <h2 className="mt-6 font-display text-3xl text-[#51452F]">
              Your wishlist is waiting
            </h2>
            <p className="mt-3 max-w-md text-sm leading-6 text-ink/60">
              Save the sarees you love and keep your favourite styles close.
            </p>
            <Link
              to="/shop"
              className="mt-8 inline-flex items-center gap-3 bg-[#75633D] px-7 py-4 text-xs font-semibold uppercase tracking-[0.15em] text-white transition hover:bg-[#51452F]"
            >
              Explore collection <ArrowRight size={16} />
            </Link>
          </div>
        ) : (
          <div className="grid grid-cols-2 gap-x-4 gap-y-8 sm:grid-cols-3 sm:gap-x-6 sm:gap-y-10 lg:grid-cols-4">

const WISHLIST_KEY = "adya-wishlist";

function readWishlist() {
  try {
    const items = JSON.parse(localStorage.getItem(WISHLIST_KEY) || "[]");
    return Array.isArray(items) ? items : [];
  } catch {
    return [];
  }
}

function toggleWishlistProduct(product) {
  const items = readWishlist();
  const exists = items.some((item) => item.id === product.id);
  const updated = exists
    ? items.filter((item) => item.id !== product.id)
    : [...items, product];

  try {
    localStorage.setItem(WISHLIST_KEY, JSON.stringify(updated));
    return !exists;
  } catch {
    return exists;
  }
}

function formatProduct(product) {
  return {
    ...product,

    priceInr: product.price,
    compareAtInr: product.compareAtPrice,
    image: product.imagePath || product.images?.[0] || "",
    to: `/product/${product.slug || product._id}`,
  };
}

import { createContext, useCallback, useEffect, useMemo, useState } from "react";

const CartContext = createContext(null);
const CART_STORAGE_KEY = "adya-cart";

function createLineId(product) {
  return [
    product.productId || product.id,
    product.variantId || "no-variant",
    product.sizeOptionId || product.selectedSize || "no-size",
  ].join(":");
}

function readCart() {
  try {
    const saved = localStorage.getItem(CART_STORAGE_KEY);
    const parsed = saved ? JSON.parse(saved) : [];

    if (!Array.isArray(parsed)) return [];

    return parsed
      .filter((line) => line && line.product)
      .map((line) => {
        const product = line.product;
        const lineId = line.lineId || createLineId({
          ...product,
          productId: line.productId || product.productId || product.id,
        });

        return {
          ...line,
          lineId,
          productId: line.productId || product.productId || product.id,
          quantity: Math.max(1, Number(line.quantity) || 1),
          maxStock: Math.max(0, Number(product.stock ?? line.maxStock ?? 0)),
        };
      });
  } catch {
    return [];
  }
}

export function CartProvider({ children }) {
  const [lines, setLines] = useState(readCart);

  useEffect(() => {
    try {
      localStorage.setItem(CART_STORAGE_KEY, JSON.stringify(lines));
    } catch {
      return;
    }
  }, [lines]);

  const addToCart = useCallback((product, quantity = 1) => {
    if (!product || !(product.productId || product.id)) return;

    const lineId = createLineId(product);
    const safeQuantity = Math.max(1, Number(quantity) || 1);
    const maxStock = Math.max(0, Number(product.stock) || 0);

    if (maxStock <= 0) return;

    setLines((current) => {
      const existing = current.find((line) => line.lineId === lineId);

      if (existing) {
        return current.map((line) =>
          line.lineId === lineId
            ? {
                ...line,
                quantity: Math.min(
                  maxStock,
                  Number(line.quantity || 0) + safeQuantity
                ),
                maxStock,
                product,
              }
            : line
        );
      }

      return [
        ...current,
        {
          lineId,
          productId: product.productId || product.id,
          variantId: product.variantId || null,
          sizeOptionId: product.sizeOptionId || null,
          quantity: Math.min(safeQuantity, maxStock),
          maxStock,
          product,
        },
      ];
    });
  }, []);

  const removeFromCart = useCallback((lineIdOrProductId) => {
    setLines((current) =>
      current.filter(
        (line) =>
          line.lineId !== lineIdOrProductId &&
          line.productId !== lineIdOrProductId
      )
    );
  }, []);

  const updateQuantity = useCallback((lineId, quantity) => {
    const nextQuantity = Number(quantity);

    if (!Number.isFinite(nextQuantity) || nextQuantity <= 0) {
      setLines((current) => current.filter((line) => line.lineId !== lineId));
      return;
    }

    setLines((current) =>
      current.map((line) =>
        line.lineId === lineId
          ? {
              ...line,
              quantity: Math.min(nextQuantity, Number(line.maxStock || 0)),
            }
          : line
      )
    );
  }, []);

  const clearCart = useCallback(() => setLines([]), []);

  const itemCount = useMemo(
    () => lines.reduce((total, line) => total + Number(line.quantity || 0), 0),
    [lines]
  );

  const value = useMemo(
    () => ({
      lines,
      itemCount,
      addToCart,
      removeFromCart,
      updateQuantity,
      clearCart,
    }),
    [lines, itemCount, addToCart, removeFromCart, updateQuantity, clearCart]
  );

  return <CartContext.Provider value={value}>{children}</CartContext.Provider>;
}

export default CartContext;

import { useEffect, useRef } from 'react';
import { createPortal } from 'react-dom';
import { Check, X } from 'lucide-react';
import useCurrency from '../../hooks/useCurrency';
import useScrollLock from '../../hooks/useScrollLock';
import { RATES_DISCLAIMER } from '../../config/currency';

export default function CountryModal({ onClose }) {
  const { country, countries, setCountryCode } = useCurrency();
  const panelRef = useRef(null);
  useScrollLock(true);

  useEffect(() => {
    const selected = panelRef.current?.querySelector('[aria-checked="true"]');
    selected?.focus();
  }, []);

  const handleKeyDown = (event) => {
    if (event.key === 'Escape') {
      event.preventDefault();
      onClose();
      return;
    }
    if (event.key !== 'Tab' || !panelRef.current) return;
    const focusable = panelRef.current.querySelectorAll('button:not([disabled])');
    if (focusable.length === 0) return;
    const first = focusable[0];
    const last = focusable[focusable.length - 1];
    if (event.shiftKey && document.activeElement === first) {
      event.preventDefault();
      last.focus();
    } else if (!event.shiftKey && document.activeElement === last) {
      event.preventDefault();
      first.focus();
    }
  };

  const handleSelect = (code) => {
    setCountryCode(code);
    onClose();
  };

  return createPortal(
    <div
      className="fixed inset-0 z-[60] flex items-end justify-center p-3 sm:items-center sm:p-6"
      onKeyDown={handleKeyDown}
    >
      <button
        type="button"
        tabIndex={-1}
        aria-label="Close country selector"
        onClick={onClose}
        className="absolute inset-0 cursor-default bg-ink/60 backdrop-blur-sm"
      />
      <div
        ref={panelRef}
        role="dialog"
        aria-modal="true"
        aria-labelledby="country-modal-title"
        className="relative flex max-h-[88vh] w-full max-w-md animate-fade-in flex-col overflow-hidden rounded-2xl border border-gold/50 bg-ivory-100/95 shadow-glass backdrop-blur-xl"
      >
        <div className="flex items-start justify-between gap-4 px-5 pb-3 pt-5">
          <div>
            <h2 id="country-modal-title" className="font-display text-2xl font-semibold text-burgundy">
              Choose your destination
            </h2>
            <p className="mt-1 text-sm text-ink/70">Prices will switch to your destination currency.</p>
          </div>
          <button type="button" onClick={onClose} aria-label="Close" className="icon-btn -mr-2 -mt-1 shrink-0">
            <X size={20} aria-hidden="true" />
          </button>
        </div>

        <div role="radiogroup" aria-labelledby="country-modal-title" className="overflow-y-auto px-3 pb-3">
          {countries.map((option) => {
            const selected = option.code === country.code;
            return (
              <button
                key={option.code}
                type="button"
                role="radio"
                aria-checked={selected}
                onClick={() => handleSelect(option.code)}
                className={`flex w-full items-center gap-3 rounded-xl px-3 py-3 text-left transition-colors ${
                  selected ? 'bg-burgundy/10' : 'hover:bg-burgundy/5'
                }`}
              >
                <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-burgundy text-xs font-semibold tracking-wider text-gold-light">
                  {option.code}
                </span>
                <span className="flex min-w-0 flex-1 flex-col">
                  <span className="truncate text-base font-medium text-ink">{option.name}</span>
                  <span className="text-xs text-ink/60">
                    {option.currency} &middot; {option.locale}
                  </span>
                </span>
                {selected && <Check size={20} className="shrink-0 text-burgundy" aria-hidden="true" />}
              </button>
            );
          })}
        </div>

        <p className="border-t border-burgundy/10 px-5 py-3 text-xs leading-relaxed text-ink/60">
          {RATES_DISCLAIMER}
        </p>
      </div>
    </div>,
    document.body,
  );
}

import { useCallback, useRef, useState } from 'react';
import { ChevronDown, Globe } from 'lucide-react';
import useCurrency from '../../hooks/useCurrency';
import CountryModal from './CountryModal';

const TRIGGER_STYLES = {
  header:
    'h-10 gap-1.5 rounded-full border border-burgundy/25 bg-white/40 px-3 text-xs font-semibold tracking-wide text-ink hover:bg-white/70',
  drawer:
    'w-full justify-between gap-3 rounded-xl border border-burgundy/20 bg-white/50 px-4 py-3 text-sm font-medium text-ink hover:bg-white/80',
  footer:
    'gap-2 rounded-full border border-ivory-100/30 bg-ivory-100/10 px-4 py-3 text-sm text-ivory-100 hover:bg-ivory-100/20',
};

export default function CountrySelector({ variant = 'header' }) {
  const { country } = useCurrency();
  const [open, setOpen] = useState(false);
  const triggerRef = useRef(null);

  const close = useCallback(() => {
    setOpen(false);
    triggerRef.current?.focus();
  }, []);

  return (
    <>
      <button
        ref={triggerRef}
        type="button"
        aria-haspopup="dialog"
        aria-expanded={open}
        aria-label={`Change destination and currency. Current: ${country.name}, ${country.currency}`}
        onClick={() => setOpen(true)}
        className={`inline-flex items-center transition-colors ${TRIGGER_STYLES[variant]}`}
      >
        <span className="inline-flex items-center gap-2">
          <Globe size={16} aria-hidden="true" />
          {variant === 'header' ? (
            <span>
              {country.code}
              <span className="hidden text-ink/60 sm:inline"> &middot; {country.currency}</span>
            </span>
          ) : (
            <span>
              {country.name} ({country.currency})
            </span>
          )}
        </span>
        <ChevronDown size={14} aria-hidden="true" />
      </button>
      {open && <CountryModal onClose={close} />}
    </>
  );
}

import { useId, useState } from 'react';
import { Send } from 'lucide-react';

const EMAIL_PATTERN = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;

const MESSAGES = {
  empty: 'Please enter your email address.',
  invalid: 'Please enter a valid email address.',
  success:
    'Thank you. Your email looks valid. Mailing list signup is not connected yet, so nothing was saved.',
};

export default function NewsletterForm({ className = '' }) {
  const [email, setEmail] = useState('');
  const [status, setStatus] = useState('idle');
  const inputId = useId();
  const messageId = useId();
  const hasError = status === 'empty' || status === 'invalid';

  const handleChange = (event) => {
    setEmail(event.target.value);
    if (status !== 'idle') setStatus('idle');
  };

  const handleSubmit = (event) => {
    event.preventDefault();
    const value = email.trim();
    if (!value) {
      setStatus('empty');
      return;
    }
    if (!EMAIL_PATTERN.test(value)) {
      setStatus('invalid');
      return;
    }
    setStatus('success');
    setEmail('');
  };

  return (
    <form noValidate onSubmit={handleSubmit} className={className}>
      <label htmlFor={inputId} className="sr-only">
        Email address
      </label>
      <div className="flex flex-col gap-3 sm:flex-row">
        <input
          id={inputId}
          type="email"
          name="email"
          autoComplete="email"
          value={email}
          onChange={handleChange}
          placeholder="Your email address"
          aria-invalid={hasError}
          aria-describedby={status === 'idle' ? undefined : messageId}
          className="h-12 min-w-0 flex-1 rounded-full border border-ivory-100/30 bg-ivory-100/10 px-5 text-base text-ivory-100 placeholder:text-ivory-100/60"
        />
        <button
          type="submit"
          className="inline-flex h-12 items-center justify-center gap-2 rounded-full bg-gold px-6 text-sm font-semibold uppercase tracking-wider text-ink transition-colors hover:bg-gold-light"
        >
          Subscribe
          <Send size={16} aria-hidden="true" />
        </button>
      </div>
      <p
        id={messageId}
        role={hasError ? 'alert' : 'status'}
        className={`mt-3 min-h-[1.25rem] text-sm ${hasError ? 'text-[#ffb4a8]' : 'text-gold-light'}`}
      >
        {status === 'idle' ? '' : MESSAGES[status]}
      </p>
    </form>
  );
}

import { useEffect, useState } from 'react';
import SocialLinks from '../common/SocialLinks';
import useCurrency from '../../hooks/useCurrency';

export default function AnnouncementBar({ messages, intervalMs = 5000 }) {
  const [index, setIndex] = useState(0);
  const [paused, setPaused] = useState(false);
  const { formatPrice } = useCurrency();
  const total = messages.length;

  useEffect(() => {
    if (paused || total < 2) return undefined;
    const timer = setInterval(() => setIndex((current) => (current + 1) % total), intervalMs);
    return () => clearInterval(timer);
  }, [paused, total, intervalMs]);

  if (total === 0) return null;
  const message = messages[index % total];

  return (
    <div
      role="region"
      aria-label="Announcements"
      className="glass-announcement"
      onMouseEnter={() => setPaused(true)}
      onMouseLeave={() => setPaused(false)}
      onFocus={() => setPaused(true)}
      onBlur={() => setPaused(false)}
    >
      <div className="mx-auto flex min-h-9 max-w-7xl items-center justify-center gap-4 px-4 py-1.5 sm:px-6 md:justify-between lg:px-8">
        <p
          key={message.id}
          className="animate-fade-in text-center text-[11px] leading-snug tracking-wide sm:text-xs md:text-left md:text-[13px]"
        >
          {message.text}
          {message.amountInr ? (
            <>
              {' '}
              <strong className="font-semibold text-gold-light">{formatPrice(message.amountInr)}</strong>
            </>
          ) : null}
        </p>
        <SocialLinks
          className="-my-1 hidden shrink-0 md:flex"
          itemClassName="text-ivory-100/80 hover:bg-ivory-100/10 hover:text-gold-light"
        />
      </div>

  );
}

import { useEffect, useRef, useState } from 'react';

import { Search, X } from 'lucide-react';
import useDismiss from '../../hooks/useDismiss';

export default function SearchPanel({ open, onClose }) {
  const [query, setQuery] = useState('');
  const inputRef = useRef(null);
  const panelRef = useRef(null);
  const navigate = useNavigate();
  useDismiss(panelRef, open, onClose, '[data-search-toggle]');

  useEffect(() => {
    if (open) inputRef.current?.focus();
  }, [open]);

  if (!open) return null;

  const handleSubmit = (event) => {

    if (!value) {

      return;

    navigate(`/shop?q=${encodeURIComponent(value)}`);
    setQuery('');
    onClose();

    <div
      ref={panelRef}
      id="header-search-panel"

import { useCallback, useMemo, useState } from 'react';
import CartContext from './CartContext';
import { INITIAL_CART_LINES } from '../data/sampleProduct';

const MAX_QUANTITY = 9;

export default function CartProvider({ children }) {
  const [lines, setLines] = useState(INITIAL_CART_LINES);

  const updateQuantity = useCallback((productId, delta) => {
    setLines((current) =>
      current.map((line) =>
        line.productId === productId
          ? { ...line, quantity: Math.min(MAX_QUANTITY, Math.max(1, line.quantity + delta)) }
          : line,
      ),
    );
  }, []);

  const itemCount = useMemo(
    () => lines.reduce((total, line) => total + line.quantity, 0),
    [lines],
  );

  const value = useMemo(() => ({ lines, itemCount, updateQuantity }), [lines, itemCount, updateQuantity]);

  return <CartContext.Provider value={value}>{children}</CartContext.Provider>;
}

========== FILE: src\context\CurrencyProvider.jsx ==========

import { useCallback, useMemo, useState } from 'react';
import CurrencyContext from './CurrencyContext';
import {
  COUNTRIES,
  DEFAULT_COUNTRY_CODE,
  STORAGE_KEY,
  getCountry,
  isSupportedCountry,
} from '../config/currency';
import { convertFromInr, formatCurrency, formatPrice } from '../utils/formatPrice';

const readStoredCode = () => {
  try {
    const stored = window.localStorage.getItem(STORAGE_KEY);
    return isSupportedCountry(stored) ? stored : DEFAULT_COUNTRY_CODE;
  } catch (error) {
    return DEFAULT_COUNTRY_CODE;
  }
};

export default function CurrencyProvider({ children }) {
  const [code, setCode] = useState(readStoredCode);
  const country = useMemo(() => getCountry(code), [code]);

  const setCountryCode = useCallback((nextCode) => {
    if (!isSupportedCountry(nextCode)) return;
    setCode(nextCode);
    try {
      window.localStorage.setItem(STORAGE_KEY, nextCode);
    } catch (error) {
      return;
    }
  }, []);

  const value = useMemo(
    () => ({
      country,
      countries: COUNTRIES,
      setCountryCode,
      convert: (amountInr) => convertFromInr(amountInr, country),
      format: (amount) => formatCurrency(amount, country),
      formatPrice: (amountInr) => formatPrice(amountInr, country),
    }),
    [country, setCountryCode],
  );

  return <CurrencyContext.Provider value={value}>{children}</CurrencyContext.Provider>;
}

import { useEffect } from 'react';

export default function useDismiss(ref, active, onDismiss, ignoreSelector = null) {
  useEffect(() => {
    if (!active) return undefined;

    const handlePointer = (event) => {
      if (ignoreSelector && event.target.closest?.(ignoreSelector)) return;
      if (ref.current && !ref.current.contains(event.target)) onDismiss();
    };
    const handleKey = (event) => {
      if (event.key === 'Escape' && !event.defaultPrevented) onDismiss();
    };

    document.addEventListener('pointerdown', handlePointer);
    document.addEventListener('keydown', handleKey);
    return () => {
      document.removeEventListener('pointerdown', handlePointer);
      document.removeEventListener('keydown', handleKey);
    };
  }, [ref, active, onDismiss, ignoreSelector]);
}

export default function useScrollLock(active) {
  useEffect(() => {
    if (!active) return undefined;
    const previous = document.body.style.overflow;
    document.body.style.overflow = 'hidden';
    return () => {
      document.body.style.overflow = previous;
    };
  }, [active]);
}

========== FILE: src\utils\formatPrice.js ==========

const formatters = new Map();

const getFormatter = (locale, currency) => {
  const key = `${locale}|${currency}`;
  if (!formatters.has(key)) {
    formatters.set(
      key,
      new Intl.NumberFormat(locale, {
        style: 'currency',
        currency,
        minimumFractionDigits: 0,
        maximumFractionDigits: 0,
      }),
    );
  }
  return formatters.get(key);
};

export const convertFromInr = (amountInr, country) => Math.round(amountInr * country.rate);

export const formatCurrency = (amount, country) =>
  getFormatter(country.locale, country.currency).format(amount);

export const formatPrice = (amountInr, country) =>
  formatCurrency(convertFromInr(amountInr, country), country);
